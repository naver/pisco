# ============================ 1. BUILD THE MIX ============================
# scripts/build_arc_mix.py is committed. Weights are relative (sum here = 100). Each source
# contributes round(w/sum * total) rows, sampled WITH replacement if the pool is smaller.
# Order of pairs is irrelevant. Verified: these weights reproduce multitask_v1 exactly
# (kilt 35000 / wikisum 18000 / regen:drop 8000 / dialogsum 7000 / regen:adversarialqa 6000 /
#  regen:freebaseqa 6000 / regen:msmarco 6000 / samsum 5000 / regen:sciq 5000 / regen:asqa 4000).

SPEC="kilt:35,regen:freebaseqa:6,regen:msmarco:6,regen:adversarialqa:6,regen:drop:8,regen:sciq:5,regen:asqa:4,wikisum:18,dialogsum:7,samsum:5"

# build_arc_mix.py now takes --zip and --regen_dir (default to the NLE paths) so you do NOT
# have to edit the script on NCP — point them at your local copies of the artifacts below.
#
# 100K build (multitask_v1):
#   python scripts/build_arc_mix.py --spec "$SPEC" --total 100000 --seed 0 \
#     --zip <NCP>/ARC-Encoder_ft.zip --regen_dir <NCP>/regen \
#     --out <ARC_FT_DIR>/multitask_v1
# 50K build (multitask_50k — the 35.7 model's data): same but --total 50000 --out .../multitask_50k
#
# ---- NCP RUNBOOK (pulling the repo is NOT enough — you need the two data artifacts) ----
#   1. git pull pisco `arc`  (gets build_arc_mix.py + this file).
#   2. ARC zip: download kyutai/ARC_finetuning from HF -> ARC-Encoder_ft.zip; check md5 vs the
#      one below. If it differs, use NLE's copy. (Provides wikisum/dialogsum/samsum.)
#   3. regen/ (193 MB, 6 files): CANNOT be regenerated (stochastic) and is NOT in git ->
#      transfer from NLE (see SYNC.md for the agreed channel). Verify the md5s below.
#   4. build (command above) with --zip/--regen_dir pointing at 2 & 3.
#   5. finetune your backbone with the recipe in section 2 (swap model_name_or_path).
#   6. eval LongBench; compare vs your single-task 32.0 and this ref (33.9 rate16 / 35.7 rate8).
#
# --seed : 0 (build_arc_mix.py default; not otherwise recorded, seed does not change
#          proportions, only WHICH rows are sampled — assume 0 for bit-for-bit rows).
# POSTPROCESSING AFTER build_arc_mix.py: NONE. No filtering / dedup / truncation. The script
#   shuffles and save_to_disk(DatasetDict{"train"}); train.py reads it directly.

# ============================ 2. FINETUNE RECIPE ============================
# Launcher used: launchers/ft_bidi_data.sh  (NOTE: *.sh is gitignored in this repo, hence this
# file). It runs `python train.py --config-name=finetune` with the overrides below. Ground truth
# = expQ/{158433,161788}/training_config.yaml. The two ref runs are IDENTICAL except dataset /
# samples / compr_rate.
#
#   ft_bidi_data.sh <pretrain_jobid> <dataset_path> <samples> <cml=128> <rate>
#   e.g. rate8 35.7 recipe:  ft_bidi_data.sh 149314 <ARC_FT_DIR>/multitask_50k 50000 128 8
#        rate16 33.9 recipe:  ft_bidi_data.sh 149314 <ARC_FT_DIR>/multitask_v1  100000 128 16
#
# config-name           : finetune
# model_name_or_path    : expQ/149314/model   (4B->4B BIDIRECTIONAL pretrain, 500K samples;
#                          NCP: substitute your own backbone here)
# model_config_overrides.compr_rate : 8   (rate16 run: omit -> inherits ckpt's rate)
#
# hf_training (overrides finetune.yaml defaults via the launcher):
#   num_train_epochs            : 1
#   per_device_train_batch_size : 8
#   gradient_accumulation_steps : 8
#   -> EFFECTIVE BATCH          : 64   (8 x 8 x 1 GPU; launcher uses --gres=gpu:1, single process)
#   learning_rate               : 1e-4
#   warmup_steps                : 0.05
#   weight_decay                : 0.1
#   max_grad_norm               : 1.0
#   bf16                        : true
#   gradient_checkpointing      : true
#   eval_strategy/eval_steps    : steps / 400   (in-training eval only; NOT used for selection)
#   save_strategy/save_steps    : steps / 1000  (save_total_limit 5)
#
# collator_class  : pisco.collator.FineTuningCollator   (PISCO, query_dependent=False; not OSCAR)
# collator_kwargs :
#   compressor_max_length : 128   (doc chunk size; NB our backbone was pretrained at doc/cml 128.
#                                  NCP backbones are doc 256 — decide whether to match our 128 to
#                                  replicate the recipe, or 256 to match your pretrain.)
#   decoder_max_length    : 1024  (launcher bumps this from the finetune.yaml default of 512)
#   chunk_docs            : true
#   (no topk_docs override; docs are chunked, not top-k truncated)
#
# CHECKPOINT SELECTION RULE : FINAL checkpoint (last save). No load_best_model_at_end; eval was
#   run on expQ/<jobid>/model (the final weights), not a step-selected best.
